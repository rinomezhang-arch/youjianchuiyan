========================================
 一、缺失的表（库里完全没有的）—— 6张必须建
========================================

1. order_master
   现场点单主表（区别于booking预订）。iPad客人入座后开台点菜用。
   字段:
   - order_id bigint PK AUTO_INCREMENT COMMENT '订单ID'
   - order_no varchar(50) NOT NULL COMMENT '订单编号(系统生成)'
   - store_id bigint NOT NULL DEFAULT 1 COMMENT '门店ID'
   - table_id int NOT NULL COMMENT '桌台ID(FK→table_master)'
   - table_number varchar(10) COMMENT '桌号'
   - customer_id int COMMENT '客户ID(FK→customer_master)'
   - guest_count int DEFAULT 0 COMMENT '用餐人数'
   - order_date date NOT NULL COMMENT '下单日期'
   - order_time time COMMENT '下单时间'
   - order_status varchar(20) DEFAULT 'ordering' COMMENT '订单状态(ordering点菜中/ordered已下单/serving上菜中/paid已结账/cancelled已取消)'
   - total_amount decimal(10,2) DEFAULT 0.00 COMMENT '原价合计'
   - discount_amount decimal(10,2) DEFAULT 0.00 COMMENT '折扣金额'
   - final_amount decimal(10,2) DEFAULT 0.00 COMMENT '实收金额'
   - payment_method varchar(20) COMMENT '支付方式(cash/wechat/alipay/card)'
   - payment_status varchar(20) DEFAULT 'unpaid' COMMENT '支付状态(unpaid/paid/refunded)'
   - staff_id int COMMENT '服务员ID(FK→staff_master)'
   - remark text COMMENT '备注'
   - created_at timestamp DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间'
   - updated_at timestamp DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间'
   外键: store_id→store_info, table_id→table_master, customer_id→customer_master, staff_id→staff_master

2. order_dish_detail
   现场点单的每道菜明细。
   字段:
   - detail_id bigint PK AUTO_INCREMENT COMMENT '明细ID'
   - order_id bigint NOT NULL COMMENT '订单ID(FK→order_master)'
   - store_id bigint NOT NULL DEFAULT 1 COMMENT '门店ID'
   - dish_id varchar(20) NOT NULL COMMENT '菜品ID(FK→dish_master)'
   - dish_name varchar(100) COMMENT '菜品名称'
   - dish_quantity int DEFAULT 1 COMMENT '点单数量'
   - unit_price decimal(10,2) COMMENT '单价'
   - subtotal decimal(10,2) COMMENT '小计(数量×单价)'
   - dish_note varchar(255) COMMENT '口味备注(少辣/免葱等)'
   - kitchen_status varchar(20) DEFAULT 'pending' COMMENT '厨房状态(pending未提交/submitted已提交/cooking制作中/done已完成/served已上菜)'
   - kitchen_station varchar(50) COMMENT '出菜口(热菜/凉菜/面点/蒸菜/烧腊)'
   - kitchen_started_at datetime COMMENT '开始制作时间'
   - kitchen_done_at datetime COMMENT '完成制作时间'
   - served_at datetime COMMENT '上菜时间'
   - created_at timestamp DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间'
   外键: order_id→order_master, dish_id→dish_master

3. order_bill
   结账单。一次用餐对应一张账单。
   字段:
   - bill_id bigint PK AUTO_INCREMENT COMMENT '账单ID'
   - bill_no varchar(50) NOT NULL COMMENT '账单编号'
   - store_id bigint NOT NULL DEFAULT 1 COMMENT '门店ID'
   - order_id bigint COMMENT '订单ID(FK→order_master,现场点单)'
   - booking_id varchar(20) COMMENT '预订ID(FK→booking_master,宴会预订)'
   - customer_id int COMMENT '客户ID(FK→customer_master)'
   - table_id int COMMENT '桌台ID(FK→table_master)'
   - total_amount decimal(10,2) NOT NULL COMMENT '原价合计'
   - discount_amount decimal(10,2) DEFAULT 0.00 COMMENT '折扣金额'
   - round_amount decimal(10,2) DEFAULT 0.00 COMMENT '抹零金额'
   - final_amount decimal(10,2) NOT NULL COMMENT '实收金额'
   - payment_method varchar(20) COMMENT '支付方式(cash现金/wechat微信/alipay支付宝/card刷卡/combined组合支付)'
   - payment_status varchar(20) DEFAULT 'unpaid' COMMENT '支付状态(unpaid/paid/refunded/partial)'
   - paid_at datetime COMMENT '收款时间'
   - staff_id int COMMENT '收银员ID(FK→staff_master)'
   - remark varchar(500) COMMENT '备注'
   - created_at timestamp DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间'
   外键: order_id→order_master, booking_id→booking_master, customer_id→customer_master, table_id→table_master, staff_id→staff_master

4. kds_queue
   厨房显示系统队列。iPad下单后菜品进此队列，后厨按屏显做菜。
   字段:
   - queue_id bigint PK AUTO_INCREMENT COMMENT '队列ID'
   - store_id bigint NOT NULL DEFAULT 1 COMMENT '门店ID'
   - source_type varchar(20) NOT NULL COMMENT '来源类型(booking预订菜品/order现场点单)'
   - source_detail_id bigint NOT NULL COMMENT '来源明细ID(booking_dish_detail.dish_booking_id 或 order_dish_detail.detail_id)'
   - dish_name varchar(100) NOT NULL COMMENT '菜品名称'
   - quantity int NOT NULL DEFAULT 1 COMMENT '数量'
   - table_number varchar(10) COMMENT '桌号'
   - station varchar(50) NOT NULL COMMENT '出菜口(热菜/凉菜/面点/蒸菜/烧腊/汤羹)'
   - priority int DEFAULT 0 COMMENT '优先级(0普通/1加急/2超时催菜)'
   - queue_status varchar(20) DEFAULT 'waiting' COMMENT '队列状态(waiting等待/cooking制作中/done已完成/cancelled已取消)'
   - order_time datetime NOT NULL COMMENT '下单时间'
   - started_at datetime COMMENT '开始制作时间'
   - done_at datetime COMMENT '完成时间'
   - remark varchar(200) COMMENT '备注(口味/特殊要求)'
   - created_at timestamp DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间'
   索引: idx_store(source_type,store_id), idx_station(station,queue_status)

5. banquet_hall
   宴会厅。预订宴会时必须选宴会厅。
   字段:
   - hall_id int PK AUTO_INCREMENT COMMENT '宴会厅ID'
   - store_id bigint NOT NULL DEFAULT 1 COMMENT '门店ID'
   - hall_name varchar(50) NOT NULL COMMENT '宴会厅名称(如:一楼龙凤厅)'
   - hall_code varchar(20) NOT NULL COMMENT '宴会厅编码'
   - floor varchar(20) COMMENT '所在楼层'
   - max_tables int NOT NULL COMMENT '最大桌数'
   - max_guests int COMMENT '最大容纳人数(桌数×每桌人数)'
   - has_stage tinyint DEFAULT 0 COMMENT '是否有舞台(0无/1有)'
   - has_led_screen tinyint DEFAULT 0 COMMENT '是否有LED屏'
   - has_sound_system tinyint DEFAULT 0 COMMENT '是否有音响系统'
   - daily_rent decimal(10,2) DEFAULT 0.00 COMMENT '日租金'
   - description varchar(500) COMMENT '描述'
   - status varchar(20) DEFAULT 'available' COMMENT '状态(available可用/maintenance维护中/closed关闭)'
   - created_at timestamp DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间'
   - updated_at timestamp DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间'
   外键: store_id→store_info

6. menu_self_order_config
   iPad自选菜单配置。
   字段:
   - config_id int PK AUTO_INCREMENT COMMENT '配置ID'
   - store_id bigint NOT NULL DEFAULT 1 COMMENT '门店ID'
   - is_enabled tinyint DEFAULT 1 COMMENT '是否启用iPad自选(0否/1是)'
   - show_recommend tinyint DEFAULT 1 COMMENT '是否显示推荐菜品(0否/1是)'
   - show_tags tinyint DEFAULT 1 COMMENT '是否显示菜品标签(0否/1是)'
   - show_price tinyint DEFAULT 1 COMMENT '是否显示价格(0否/1是)'
   - enable_custom_note tinyint DEFAULT 1 COMMENT '是否允许口味备注(0否/1是)'
   - menu_style varchar(20) DEFAULT 'grid' COMMENT '菜单样式(grid网格/list列表)'
   - created_at timestamp DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间'
   - updated_at timestamp DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间'
