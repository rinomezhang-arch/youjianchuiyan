### member_consume_record 建表语句

*************************** 1. row ***************************
       Table: member_consume_record
Create Table: CREATE TABLE `member_consume_record` (
  `consume_id` bigint NOT NULL AUTO_INCREMENT COMMENT '消费ID',
  `store_id` bigint NOT NULL DEFAULT '1' COMMENT '门店ID',
  `consume_no` varchar(50) COLLATE utf8mb4_unicode_ci NOT NULL COMMENT '消费单号',
  `member_id` bigint NOT NULL COMMENT '会员ID',
  `card_no` varchar(50) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '会员卡号',
  `member_name` varchar(100) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '会员姓名',
  `consume_date` date NOT NULL COMMENT '消费日期',
  `booking_id` int DEFAULT NULL COMMENT '预订ID',
  `booking_no` varchar(50) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '预订编号',
  `consume_amount` decimal(12,2) NOT NULL COMMENT '消费金额',
  `discount_amount` decimal(12,2) DEFAULT '0.00' COMMENT '优惠金额',
  `actual_amount` decimal(12,2) NOT NULL COMMENT '实付金额',
  `balance_pay` decimal(12,2) DEFAULT '0.00' COMMENT '余额支付',
  `cash_pay` decimal(12,2) DEFAULT '0.00' COMMENT '现金支付',
  `other_pay` decimal(12,2) DEFAULT '0.00' COMMENT '其他支付',
  `balance_before` decimal(12,2) DEFAULT NULL COMMENT '消费前余额',
  `balance_after` decimal(12,2) DEFAULT NULL COMMENT '消费后余额',
  `points_earned` int DEFAULT '0' COMMENT '获得积分',
  `points_used` int DEFAULT '0' COMMENT '抵扣积分',
  `operator_id` int DEFAULT NULL COMMENT '操作人ID',
  `operator_name` varchar(50) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '操作人姓名',
  `remark` varchar(500) COLLATE utf8mb4_unicode_ci DEFAULT NULL COMMENT '备注',
  `created_at` timestamp NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
  PRIMARY KEY (`consume_id`),
  KEY `idx_store_id` (`store_id`),
  KEY `idx_consume_no` (`consume_no`),
  KEY `idx_member_id` (`member_id`),
  KEY `idx_consume_date` (`consume_date`),
  KEY `idx_booking_id` (`booking_id`)
) ENGINE=InnoDB AUTO_INCREMENT=41 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='会员消费记录表'
